SPARQL Query

PREFIX s223: 
PREFIX qudt: 
PREFIX quantitykind: 

SELECT DISTINCT ?sensorLabel
WHERE {
    ?sensor a/rdfs:subClassOf* s223:Sensor .
    ?sensor s223:observes ?prop .
    {
       ?prop qudt:hasQuantityKind quantitykind:Temperature .
    }
    UNION
    {
       ?prop qudt:unit/qudt:hasQuantityKind quantitykind:Temperature .
    }
    ?sensor rdfs:label ?sensorLabel .
}
