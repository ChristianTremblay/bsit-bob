SPARQL Query

4-average-temp-of-something

PREFIX s223: 
PREFIX qudt: 
PREFIX qudtqk: 

SELECT ?roomLabel (AVG(?tval) AS ?avg)
WHERE {
    ?oprop a/rdfs:subClassOf* s223:Property .
    ?oprop s223:hasValue s223:OccupancyStatus-Unoccupied .
    ?ds s223:hasProperty ?oprop .
    ?room s223:encloses ?ds .
    ?room s223:encloses ?otherds .
    ?otherds s223:hasProperty ?tprop .
    ?tprop qudt:hasQuantityKind qudtqk:Temperature .
    ?tprop s223:hasValue ?tval .
    ?room rdfs:label ?roomLabel .
}
GROUP BY ?roomLabel
