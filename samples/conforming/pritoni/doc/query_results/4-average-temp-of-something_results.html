PREFIX s223: <http://data.ashrae.org/standard223#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX qudtqk: <http://qudt.org/vocab/quantitykind/>

SELECT ?sensorLabel ?extref
WHERE {
    ?oprop a/rdfs:subClassOf* s223:Property .
    ?sensor s223:observes ?oprop .
    ?oprop qudt:hasQuantityKind qudtqk:Temperature .
    ?oprop s223:hasExternalReference ?extref .
    ?sensor rdfs:label ?sensorLabel .
}
GROUP BY ?sensorLabel
