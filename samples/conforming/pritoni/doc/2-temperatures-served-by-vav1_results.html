SPARQL Query

2-temperatures-served-by-vav1

PREFIX psm: 
PREFIX s223: 
PREFIX qudt: 
PREFIX qudtqk: 

SELECT DISTINCT ?vav ?sensor ?unit
WHERE {
    ?sensor s223:hasObservationLocation ?ds .
    ?sensor a ?sensorType .
    ?sensorType rdfs:subClassOf* s223:Sensor .
    ?sensor s223:observes ?t .
    ?t qudt:hasQuantityKind qudtqk:Temperature .
    ?t qudt:unit ?unit .
    ?vav s223:TerminalUnit* ?eq .
    ?eq s223:contains ?sensor .
}

GROUP BY ?vav ?sensor ?unit
