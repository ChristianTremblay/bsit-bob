2-temperatures-served-by-vav1

PREFIX psm: 
PREFIX s223: 
PREFIX qudt: 
PREFIX qudtqk: 

SELECT ?sensor ?unit
WHERE {
    SELECT DISTINCT ?sensor ?value ?unit
    WHERE {
        ?zone s223:contains ?ds .
        ?sensor s223:hasObservationLocation ?ds .
        ?sensor a/rdfs:subClassOf* s223:Sensor .
        ?sensor s223:observes ?t .
        ?t qudt:hasQuantityKind qudtqk:Temperature .
        ?t qudt:unit ?unit .
    }
}
GROUP BY ?unit
